# The symbols Wine's winex11.drv looks up in the Vulkan library it dlopens.
#
# Wine 9.0, dlls/winex11.drv/vulkan.c: the driver dlopens SONAME_LIBVULKAN and
# resolves each of these with dlsym. A name in the "required" block that the
# library does not export makes the driver's LOAD_FUNCPTR fail, it closes the
# handle, and the whole Vulkan path is gone -- which is indistinguishable, in
# a device log, from the library not being found at all. The "optional" block
# is LOAD_OPTIONAL_FUNCPTR: absent is allowed and only costs the feature.
#
# Everything else Wine needs it obtains through vkGetInstanceProcAddr and
# vkGetDeviceProcAddr, which is why the bridge's command table is much longer
# than this file.
#
# scripts/validate-x64-vulkan-shim.py --winex11 PATH re-measures a given
# winex11.so against this list by scanning its string table, so a Wine version
# that added a required symbol fails the build rather than a device run.

[required]
vkCreateInstance
vkCreateSwapchainKHR
vkCreateXlibSurfaceKHR
vkDestroyInstance
vkDestroySurfaceKHR
vkDestroySwapchainKHR
vkEnumerateInstanceExtensionProperties
vkGetDeviceProcAddr
vkGetInstanceProcAddr
vkGetPhysicalDeviceSurfaceCapabilitiesKHR
vkGetPhysicalDeviceSurfaceFormatsKHR
vkGetPhysicalDeviceSurfacePresentModesKHR
vkGetPhysicalDeviceSurfaceSupportKHR
vkGetPhysicalDeviceXlibPresentationSupportKHR
vkGetSwapchainImagesKHR
vkQueuePresentKHR

[optional]
vkGetDeviceGroupSurfacePresentModesKHR
vkGetPhysicalDevicePresentRectanglesKHR
vkGetPhysicalDeviceSurfaceCapabilities2KHR
vkGetPhysicalDeviceSurfaceFormats2KHR

# Names that appear as strings inside winex11.so but are NOT dlsym'd out of the
# Vulkan library, so the shim owes them nothing. Each is recorded with the
# reason, because a bare string scan cannot tell a dlsym argument from any other
# literal and the first CI run of this validator failed on exactly these.
#
# vkCreateWin32SurfaceKHR
#   A literal in wine_vk_host_fn_name(), which translates the Win32 spelling to
#   "vkCreateXlibSurfaceKHR" before anything is looked up. It is also the
#   driver's own exported X11DRV_vkCreateWin32SurfaceKHR. Never dlsym'd.
# vkGetPhysicalDeviceWin32PresentationSupportKHR
#   The same translation pair: mapped to
#   "vkGetPhysicalDeviceXlibPresentationSupportKHR", which *is* in [required].
# vkGetPhysicalDeviceProperties2KHR
# vkGetPhysicalDeviceMemoryProperties2KHR
#   Absent from the LOAD_FUNCPTR list. Wine and DXVK request these through
#   vkGetInstanceProcAddr, which is why the bridge aliases them onto its core
#   *2 commands rather than exporting them as separate entry points.
# vkGetRandROutputDisplayEXT
#   Part of winex11's display-enumeration handling, resolved through
#   vkGetInstanceProcAddr if at all. Not in the LOAD_FUNCPTR list.
[not-imported]
vkCreateWin32SurfaceKHR
vkGetPhysicalDeviceMemoryProperties2KHR
vkGetPhysicalDeviceProperties2KHR
vkGetPhysicalDeviceWin32PresentationSupportKHR
vkGetRandROutputDisplayEXT
