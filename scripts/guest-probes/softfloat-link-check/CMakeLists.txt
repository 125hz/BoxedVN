cmake_minimum_required(VERSION 3.20)
project(SoftFloatLinkCheck C CXX)
get_filename_component(BOXEDVN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FEX_SOURCE "${BOXEDVN_ROOT}/third_party/fex64/fex" CACHE PATH "Pinned FEX source")
option(BVN_ISOLATE_FEX "Use the production symbol namespace" ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(namespace_header "${CMAKE_CURRENT_BINARY_DIR}/fex-softfloat.h")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${BOXEDVN_ROOT}/scripts/fex-softfloat-namespace.py"
    "${FEX_SOURCE}" "${namespace_header}" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${FEX_SOURCE}/External/SoftFloat-3e" fex-softfloat)
add_library(fex_call STATIC fex.c)
target_link_libraries(fex_call PUBLIC softfloat_3e)
add_library(fex_cpp_call STATIC fex.cpp)
target_link_libraries(fex_cpp_call PUBLIC softfloat_3e)
if(BVN_ISOLATE_FEX)
    target_compile_options(softfloat_3e PRIVATE "-include${namespace_header}")
    target_compile_options(fex_call PRIVATE "-include${namespace_header}")
    target_compile_options(fex_cpp_call PRIVATE "-include${namespace_header}")
endif()
file(GLOB legacy_sources "${BOXEDVN_ROOT}/lib/softfloat/source/*.c"
    "${BOXEDVN_ROOT}/lib/softfloat/source/8086-SSE/*.c")
add_library(legacy_softfloat STATIC ${legacy_sources})
target_include_directories(legacy_softfloat PUBLIC "${BOXEDVN_ROOT}/lib/softfloat/source/include"
    "${BOXEDVN_ROOT}/lib/softfloat/source/8086-SSE")
add_library(legacy_call STATIC legacy.c)
target_link_libraries(legacy_call PUBLIC legacy_softfloat)
add_executable(check main.c)
# Deliberately load the original library first, as the iOS link does.
target_link_libraries(check PRIVATE legacy_call legacy_softfloat fex_call fex_cpp_call softfloat_3e)
enable_testing()
add_test(NAME both_softfloat_abis COMMAND check)
